\section*{Questão 4}

\section*{Solução}

\textbf{1) Fatoração de $q$.} Como $q(-1)=-1+3-2=0$, aplicando Briot--Ruffini:
\[
q(x)=x^3-3x-2=(x+1)(x^2-x-2)=(x+1)^2(x-2).
\]
Logo $-1$ é raiz dupla de $q$ e $2$ é raiz simples.

\textbf{2) Condições de divisibilidade.} Os fatores $(x+1)^2$ e $(x-2)$ são primos entre si; portanto $q\mid p$ se, e somente se, $(x-2)\mid p$ e $(x+1)^2\mid p$, isto é,
\[
p(2)=0,\qquad p(-1)=0,\qquad p'(-1)=0,
\]
pois $-1$ é raiz de multiplicidade pelo menos $2$ de $p$ exatamente quando anula $p$ e $p'$.

\begin{itemize}
\item $p(-1)=-1+3a+1-5a-7-7a-9+a=-8a-16=0 \;\Rightarrow\; a=-2$.
\item $p(2)=128+192a-32-80a+56-28a+18+a=170+85a=0 \;\Rightarrow\; a=-2$.
\item $p'(x)=7x^6+18ax^5-5x^4-20ax^3+21x^2-14ax+9$, então
$p'(-1)=7-18a-5+20a+21+14a+9=32+16a=0 \;\Rightarrow\; a=-2$.
\end{itemize}
As três condições são compatíveis e dão o mesmo valor, $a=-2$ (o único possível). Assim,
\[
p(x)=x^7-6x^6-x^5+10x^4+7x^3+14x^2+9x-2 .
\]

\textbf{3) Divisão por $q$.} Fazendo a divisão de $p$ por $x^3-3x-2$ (método da chave), obtêm-se os quocientes parciais $x^4,\,-6x^3,\,2x^2,\,-6x,\,1$ e resto nulo:
\[
p(x)=(x^3-3x-2)\,(x^4-6x^3+2x^2-6x+1).
\]

\textbf{4) Raízes do quociente.} $Q(x)=x^4-6x^3+2x^2-6x+1$ é recíproco (coeficientes simétricos) e $Q(0)=1\neq 0$. Dividindo $Q(x)=0$ por $x^2$:
\[
\left(x^2+\frac{1}{x^2}\right)-6\left(x+\frac{1}{x}\right)+2=0 .
\]
Com $t=x+\dfrac{1}{x}$, temos $x^2+\dfrac{1}{x^2}=t^2-2$, logo
\[
t^2-2-6t+2=0\;\Longleftrightarrow\; t(t-6)=0 .
\]
\begin{itemize}
\item $t=0$: $x^2+1=0 \Rightarrow x=\pm i$.
\item $t=6$: $x^2-6x+1=0 \Rightarrow x=\dfrac{6\pm\sqrt{32}}{2}=3\pm 2\sqrt{2}$.
\end{itemize}
Conferência: $(x^2+1)(x^2-6x+1)=x^4-6x^3+2x^2-6x+1=Q(x)$.

\textbf{5) Conclusão.}
\[
p(x)=(x+1)^2(x-2)(x^2+1)(x^2-6x+1),
\]
cujas raízes são $-1$ (dupla), $2$, $i$, $-i$, $3+2\sqrt{2}$ e $3-2\sqrt{2}$, num total de $7$ raízes contadas com multiplicidade, o que confere com o grau de $p$. As raízes não reais $\pm i$ formam um par conjugado, como deve ocorrer em um polinômio de coeficientes reais.

\medskip
ANSWER: $a=-2$; raízes de $p$: $-1$ (dupla), $2$, $i$, $-i$, $3+2\sqrt{2}$ e $3-2\sqrt{2}$.
