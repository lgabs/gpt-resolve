\section*{Questão 3}

\section*{Solução}

\textbf{1. As raízes.} Como $i\neq 0$ e $\dfrac{1}{i}=-i$,
\[
r+iz^8=0 \iff z^8=-\frac{r}{i}=ri=r\left(\cos\frac{\pi}{2}+i\,\mathrm{sen}\,\frac{\pi}{2}\right).
\]
Escrevendo $z=\rho(\cos\theta+i\,\mathrm{sen}\,\theta)$ e aplicando a fórmula de De Moivre, obtemos $\rho^8=r$ e $8\theta=\frac{\pi}{2}+2k\pi$. Logo as raízes são
\[
z_k=\sqrt[8]{r}\left(\cos\theta_k+i\,\mathrm{sen}\,\theta_k\right),\qquad \theta_k=\frac{\pi}{16}+\frac{k\pi}{4},\quad k=0,1,\dots,7 .
\]
São oito números distintos, todos com o mesmo módulo $R=\sqrt[8]{r}>0$ e argumentos consecutivos separados por $\frac{2\pi}{8}=\frac{\pi}{4}$. Portanto, ligados em ordem crescente de argumento, eles são os vértices de um \textbf{octógono regular} inscrito na circunferência de centro $0$ e raio $R$.

\textbf{2. A área e o valor de $r$.} Os segmentos que ligam a origem aos vértices dividem o octógono em $8$ triângulos isósceles congruentes. Cada um tem dois lados iguais a $R$ e ângulo $\frac{\pi}{4}$ entre eles. Assim,
\[
S=8\cdot\frac{1}{2}R^2\,\mathrm{sen}\,\frac{\pi}{4}=8\cdot\frac12 R^2\cdot\frac{\sqrt2}{2}=2\sqrt2\,R^2 .
\]
Com $S=\sqrt2$, temos $2\sqrt2\,R^2=\sqrt2$, ou seja, $R^2=\dfrac12$. Daí
\[
r=R^8=\left(R^2\right)^4=\left(\frac12\right)^4=\frac{1}{16}.
\]

\textbf{3. O lado.} Para o lado $\ell=|z_{k+1}-z_k|$, aplicamos a lei dos cossenos no triângulo $0\,z_k\,z_{k+1}$:
\[
\ell^2=R^2+R^2-2R^2\cos\frac{\pi}{4}=2R^2\left(1-\frac{\sqrt2}{2}\right)=1-\frac{\sqrt2}{2}=\frac{2-\sqrt2}{2}.
\]
Portanto
\[
\ell=\sqrt{\frac{2-\sqrt2}{2}}=\frac{\sqrt{4-2\sqrt2}}{2}\approx 0{,}54 .
\]

\textbf{4. Verificações.}
\begin{itemize}
\item Para um octógono regular de lado $\ell$, vale $S=2(1+\sqrt2)\ell^2$. Com o valor encontrado, $2(1+\sqrt2)\cdot\dfrac{2-\sqrt2}{2}=(1+\sqrt2)(2-\sqrt2)=2-\sqrt2+2\sqrt2-2=\sqrt2$, como no enunciado.
\item Com $r=\frac1{16}$, temos $R=\frac{\sqrt2}{2}\approx0{,}71$. A razão $\ell/R=2\,\mathrm{sen}\,\frac{\pi}{8}\approx0{,}77$ é compatível com o octógono regular.
\item Como $r>0$, a condição do enunciado é satisfeita.
\end{itemize}

\medskip
\noindent ANSWER: o polígono é um octógono regular de lado $\ell=\sqrt{\dfrac{2-\sqrt2}{2}}=\dfrac{\sqrt{4-2\sqrt2}}{2}\approx 0{,}54$, e $r=\dfrac{1}{16}$.
