\section*{Questão 9}

\section*{Solução}

\textbf{Hipóteses.} Temperatura de $25\ ^\circ$C, logo $K_w = 1\cdot10^{-14}$. Volumes aditivos. HX é ácido forte, então seu ânion $X^-$ é neutro. ZOH é base forte, então $Z^+$ também é neutro.

\textbf{1) Quantidade total de ácido.} O volume de 20 mL corresponde ao ponto de equivalência final, em que os dois ácidos já foram neutralizados. Os dois são monopróticos e reagem $1:1$ com $OH^-$:
\[
n_{HX}+n_{HY}=n_{OH^-}=0{,}1\ \text{mol/L}\times 0{,}020\ \text{L}=2{,}0\cdot10^{-3}\ \text{mol}.
\]

\textbf{2) Composição no ponto de equivalência.} A solução final, com $V = 100+20 = 120$ mL, contém apenas $Z^+$, $X^-$ e $Y^-$. Os íons $Z^+$ e $X^-$ não sofrem hidrólise. Portanto, o pH básico se deve somente à hidrólise do ânion do ácido fraco:
\[
Y^-_{(aq)} + H_2O_{(\ell)} \rightleftharpoons HY_{(aq)} + OH^-_{(aq)},\qquad
K_b=\frac{K_w}{K_a}=\frac{10^{-14}}{10^{-6}}=10^{-8}.
\]
Com pH $=9$, temos $[H^+]=10^{-9}$ e $[OH^-]=10^{-5}\ \text{mol/L}$. A hidrólise forma $HY$ e $OH^-$ na proporção $1:1$. A contribuição da água ($[H^+]=10^{-9}\ll 10^{-5}$) é desprezível, logo $[HY]\simeq[OH^-]=10^{-5}\ \text{mol/L}$. Daí:
\[
[Y^-]=\frac{[HY][OH^-]}{K_b}=\frac{10^{-5}\cdot10^{-5}}{10^{-8}}=1{,}0\cdot10^{-2}\ \text{mol/L}.
\]
A concentração analítica de Y no ponto de equivalência é
\[
C_Y=[Y^-]+[HY]=1{,}0\cdot10^{-2}+1{,}0\cdot10^{-5}\simeq 1{,}0\cdot10^{-2}\ \text{mol/L}.
\]
\emph{Validade:} a fração hidrolisada é $10^{-5}/10^{-2}=0{,}1\%$, o que justifica a aproximação.

\textbf{3) Ácido fraco HY (item b).} Pela conservação de matéria do grupo Y, a quantidade de HY é a mesma na solução inicial e no ponto de equivalência:
\[
n_{HY}=C_Y\cdot V_{\text{final}}=1{,}0\cdot10^{-2}\times 0{,}120=1{,}2\cdot10^{-3}\ \text{mol}
\;\Rightarrow\;
[HY]_0=\frac{1{,}2\cdot10^{-3}}{0{,}100}=1{,}2\cdot10^{-2}\ \text{mol/L}.
\]

\textbf{4) Ácido forte HX (item a).}
\[
n_{HX}=2{,}0\cdot10^{-3}-1{,}2\cdot10^{-3}=8{,}0\cdot10^{-4}\ \text{mol}
\;\Rightarrow\;
[HX]_0=\frac{8{,}0\cdot10^{-4}}{0{,}100}=8{,}0\cdot10^{-3}\ \text{mol/L}.
\]

\textbf{Verificação.} No ponto de equivalência, $[Z^+]=2{,}0\cdot10^{-3}/0{,}120\simeq1{,}67\cdot10^{-2}$ e $[X^-]=8{,}0\cdot10^{-4}/0{,}120\simeq6{,}7\cdot10^{-3}\ \text{mol/L}$. O balanço de cargas $[Z^+]+[H^+]=[X^-]+[Y^-]+[OH^-]$ fica $1{,}67\cdot10^{-2}\simeq 6{,}7\cdot10^{-3}+1{,}0\cdot10^{-2}$, o que é coerente. Além disso, as duas concentrações são positivas e $0{,}008+0{,}012=0{,}020\ \text{mol/L}$ é a acidez total titulada, como esperado.

\medskip
\noindent a) $[HX]=8{,}0\cdot10^{-3}\ \text{mol}\cdot\text{L}^{-1}$;\qquad b) $[HY]=1{,}2\cdot10^{-2}\ \text{mol}\cdot\text{L}^{-1}$.

\medskip
\noindent ANSWER: a) $[HX] = 8{,}0\cdot10^{-3}\ \text{mol/L}$; b) $[HY] = 1{,}2\cdot10^{-2}\ \text{mol/L}$.
