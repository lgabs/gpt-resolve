\section*{Questão 1}

\section*{Solução}

\textbf{Hipóteses.} Os reagentes estão puros, as reações são completas (rendimento de 100\%) e todo o $\text{Cl}_2$ produzido é usado na oxidação do brometo.

\textbf{1) Produção do cloro (pirolusita + HC$\ell$ concentrado).}
O Mn passa de $+4$ (em $\text{MnO}_2$) para $+2$ (em $\text{MnCl}_2$), ganhando $2e^-$. Dois íons cloreto, de $-1$, formam $\text{Cl}_2$ ($0$) e perdem $2e^-$. As semirreações são:
\[
\text{MnO}_2 + 4\,\text{H}^+ + 2e^- \longrightarrow \text{Mn}^{2+} + 2\,\text{H}_2\text{O},
\qquad
2\,\text{Cl}^- \longrightarrow \text{Cl}_2 + 2e^- .
\]
Somando as semirreações e completando com os íons espectadores, obtém-se a equação global:
\[
\text{MnO}_2(s) + 4\,\text{HCl}(aq) \xrightarrow{\ \Delta\ } \text{MnCl}_2(aq) + \text{Cl}_2(g) + 2\,\text{H}_2\text{O}(\ell)
\]
O ácido precisa ser concentrado (e a mistura costuma ser aquecida) porque a concentração elevada de $\text{H}^+$ e $\text{Cl}^-$ desloca a reação no sentido da formação de $\text{Cl}_2$. Logo, $1$ mol de $\text{MnO}_2$ fornece $1$ mol de $\text{Cl}_2$.

\textbf{2) Liberação do bromo.}
O $\text{Cl}_2$ é um oxidante mais forte que o $\text{Br}_2$ (reatividade dos halogênios: $\text{F}_2>\text{Cl}_2>\text{Br}_2>\text{I}_2$). Por isso ele desloca o bromo do brometo:
\[
\text{Cl}_2(g) + 2\,\text{NaBr}(aq) \longrightarrow 2\,\text{NaCl}(aq) + \text{Br}_2(\ell)
\]
Assim, $1$ mol de $\text{Cl}_2$ libera o bromo de $2$ mol de NaBr.

\textbf{3) Estequiometria global.}
Combinando as duas etapas:
\[
1\ \text{MnO}_2 \;\longrightarrow\; 1\ \text{Cl}_2 \;\longrightarrow\; 2\ \text{NaBr}
\quad\Longrightarrow\quad
n_{\text{MnO}_2}=\frac{n_{\text{NaBr}}}{2}.
\]

\textbf{4) Cálculos} (massas atômicas da tabela fornecida):
\[
M_{\text{NaBr}} = 22{,}990 + 79{,}904 = 102{,}894\ \text{g/mol},
\qquad
n_{\text{NaBr}} = \frac{2{,}06\ \text{g}}{102{,}894\ \text{g/mol}} \approx 2{,}002\times 10^{-2}\ \text{mol}.
\]
\[
n_{\text{Cl}_2} = n_{\text{MnO}_2} = \frac{2{,}002\times 10^{-2}}{2} \approx 1{,}001\times 10^{-2}\ \text{mol}.
\]
\[
M_{\text{MnO}_2} = 54{,}938 + 2\cdot 15{,}999 = 86{,}936\ \text{g/mol}.
\]
\[
m_{\text{MnO}_2} = 1{,}001\times 10^{-2}\ \text{mol}\cdot 86{,}936\ \text{g/mol} \approx 0{,}870\ \text{g}.
\]

\textbf{Verificação.} A ordem de grandeza confere: $2{,}06$ g de NaBr correspondem a cerca de $0{,}02$ mol. Isso exige cerca de $0{,}01$ mol de $\text{MnO}_2$, ou seja, cerca de $0{,}01\cdot 87 \approx 0{,}87$ g. O resultado tem $3$ algarismos significativos, como o dado de $2{,}06$ g. Se a pirolusita fosse impura, a massa de minério seria maior: $0{,}870\ \text{g}$ dividido pelo teor de $\text{MnO}_2$.

\textbf{Resposta.} São necessários aproximadamente $0{,}870$ g de $\text{MnO}_2$. As reações envolvidas são:
\[
\text{MnO}_2 + 4\,\text{HCl} \longrightarrow \text{MnCl}_2 + \text{Cl}_2 + 2\,\text{H}_2\text{O},
\qquad
\text{Cl}_2 + 2\,\text{NaBr} \longrightarrow 2\,\text{NaCl} + \text{Br}_2 .
\]

ANSWER: $m_{\text{MnO}_2} \approx 0{,}870\ \text{g}$, pelas reações $\text{MnO}_2 + 4\,\text{HCl} \to \text{MnCl}_2 + \text{Cl}_2 + 2\,\text{H}_2\text{O}$ e $\text{Cl}_2 + 2\,\text{NaBr} \to 2\,\text{NaCl} + \text{Br}_2$.
