\section*{Questão 10}

\section*{Solução}

\textbf{1. Modelo (relógio Th--Pb).} O decaimento radioativo segue cinética de primeira ordem. Sendo $N_0$ o número inicial de núcleos de $^{232}\text{Th}$ na formação do mineral ($t=0$):
\[
N_{\text{Th}}(t)=N_0\,e^{-\lambda t},\qquad \lambda=\frac{\ln 2}{t_{1/2}} .
\]
Toda a série do $^{232}\text{Th}$ termina em $^{208}\text{Pb}$. Os nuclídeos intermediários têm meias-vidas desprezíveis frente a $1{,}41\times10^{10}$ anos, logo sua quantidade acumulada é desprezível. Assim, cada núcleo de Th que decai origina um átomo de $^{208}\text{Pb}$. Pelas hipóteses do enunciado (sistema fechado e chumbo exclusivamente radiogênico, isto é, $^{208}\text{Pb}$ inicial nulo):
\[
N_{\text{Pb}}=N_0-N_{\text{Th}}=N_{\text{Th}}\left(e^{\lambda t}-1\right)
\;\Longrightarrow\;
R=\frac{n(^{208}\text{Pb})}{n(^{232}\text{Th})}=e^{\lambda t}-1 .
\]
Daí:
\[
t=\frac{\ln(1+R)}{\lambda}=\frac{t_{1/2}}{\ln 2}\,\ln(1+R).
\]
A razão dada já é molar (atômica). Por isso, os teores de $\text{ThO}_2$ e $\text{U}_3\text{O}_8$ e as massas molares não entram no cálculo. Os dados do urânio também não são necessários para o relógio Th--Pb.

\textbf{2. Cálculo do logaritmo com os dados fornecidos.} Com $R=0{,}024$:
\[
1+R=1{,}024=\frac{128}{125}=\frac{2^7}{5^3}
\;\Longrightarrow\;
\ln(1{,}024)=7\ln 2-3\ln 5\approx 7(0{,}693)-3(1{,}609)=4{,}851-4{,}827=0{,}024 .
\]
Esse valor é coerente com $\ln(1+x)\approx x$ para $x\ll 1$.

\textbf{3. Idade.} A constante de decaimento vale
\[
\lambda=\frac{0{,}693}{1{,}41\times10^{10}\ \text{ano}}\approx 4{,}91\times10^{-11}\ \text{ano}^{-1}.
\]
Logo:
\[
t=\frac{1{,}41\times10^{10}\ \text{ano}}{0{,}693}\times 0{,}024\approx 4{,}9\times10^{8}\ \text{anos}.
\]

\textbf{4. Verificações.}
\begin{itemize}
\item \emph{Unidades:} $\ln(1+R)$ é adimensional e $1/\lambda$ está em anos, logo $t$ sai em anos.
\item \emph{Ordem de grandeza:} $R$ é pequeno, então $t\ll t_{1/2}$. De fato, $t/t_{1/2}\approx 0{,}035$, de modo que restam cerca de $97{,}6\,\%$ do $^{232}\text{Th}$ inicial ($1/1{,}024$).
\item \emph{Sensibilidade ao arredondamento:} com logaritmos mais precisos, $\ln 1{,}024=0{,}0237$, o que dá $t\approx4{,}8\times10^{8}$ anos. A estimativa, portanto, é de cerca de 480 a 490 milhões de anos.
\item \emph{Consistência com o urânio:} se o sistema U--Pb também ficou fechado, deve-se ter $^{206}\text{Pb}/^{238}\text{U}=e^{\lambda_U t}-1$, com $\lambda_U=0{,}693/(4{,}47\times10^{9})$. Isso dá $\approx 0{,}08$, que serviria de checagem independente da idade.
\end{itemize}

ANSWER: $t=\dfrac{t_{1/2}(\text{Th})}{\ln 2}\ln(1+0{,}024)=\dfrac{1{,}41\times10^{10}}{0{,}693}\,(7\ln2-3\ln5)\approx 4{,}9\times10^{8}$ anos (cerca de 490 milhões de anos).
