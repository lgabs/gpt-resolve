\section*{Questão 9}

\section*{Solução}

\textbf{1. Elementos da elipse.} Dividindo por 16,
\[
\frac{x^2}{16}+\frac{y^2}{4}=1 \quad\Rightarrow\quad a=4,\ b=2,\ c=\sqrt{a^2-b^2}=\sqrt{12}=2\sqrt{3}.
\]
Os focos são $(\pm 2\sqrt{3},0)$. Portanto as retas pedidas são $x=\pm 2\sqrt{3}$.

\textbf{2. Normal em $P$ e ponto $Q$.} Seja $P=(x_0,y_0)$ um ponto da elipse, isto é, $x_0^2+4y_0^2=16$. Tome $F(x,y)=x^2+4y^2$. O gradiente $\nabla F(P)=(2x_0,\,8y_0)$ é perpendicular à curva em $P$, logo a normal em $P$ tem direção $(x_0,4y_0)$. Sua equação é
\[
4y_0\,(x-x_0)-x_0\,(y-y_0)=0 .
\]
Fazendo $y=0$, obtemos $4y_0(x-x_0)+x_0y_0=0$. Para $y_0\neq 0$ isso dá
\[
x=x_0-\frac{x_0}{4}=\frac{3x_0}{4}\quad\Rightarrow\quad Q=\left(\frac{3x_0}{4},\,0\right).
\]
Esse caso inclui $P=(0,\pm2)$: a normal é o eixo $y$ e $Q=(0,0)$. Já em $P=(\pm4,0)$, isto é, $y_0=0$, a normal coincide com o próprio eixo $x$. Nesse caso $Q$ não fica determinado e esses dois pontos ficam excluídos.

\textbf{3. Lugar geométrico de $M$.} O ponto médio de $PQ$ é
\[
M=(X,Y)=\left(\frac{x_0+\frac{3x_0}{4}}{2},\,\frac{y_0+0}{2}\right)=\left(\frac{7x_0}{8},\,\frac{y_0}{2}\right).
\]
Daí vêm $x_0=\dfrac{8X}{7}$ e $y_0=2Y$. Substituindo em $x_0^2+4y_0^2=16$:
\[
\frac{64X^2}{49}+16Y^2=16 \quad\Longleftrightarrow\quad \frac{X^2}{\left(\frac{7}{2}\right)^2}+\frac{Y^2}{1^2}=1 .
\]
O lugar geométrico de $M$ é, portanto, a elipse $\dfrac{4x^2}{49}+y^2=1$, de semieixos $\frac{7}{2}$ e $1$. Dela se excluem os vértices $\left(\pm\frac{7}{2},0\right)$, que viriam de $P=(\pm4,0)$.

\textbf{4. Interseção com $x=\pm c$.} Fazendo $x=\pm2\sqrt{3}$, ou seja $x^2=12$:
\[
y^2=1-\frac{4\cdot 12}{49}=\frac{1}{49}\quad\Rightarrow\quad y=\pm\frac{1}{7}.
\]
Como $c=2\sqrt{3}\approx 3{,}46<\frac{7}{2}$, cada reta corta o lugar em dois pontos. Nenhum deles é um dos vértices excluídos.

\textbf{Verificação.} Para $M=\left(2\sqrt{3},\frac{1}{7}\right)$, temos $P=\left(\frac{16\sqrt{3}}{7},\frac{2}{7}\right)$. Esse $P$ está na elipse, pois $\frac{768}{49}+\frac{16}{49}=\frac{784}{49}=16$. Além disso, $Q=\left(\frac{12\sqrt{3}}{7},0\right)$, e o ponto médio de $PQ$ é $\left(\frac{28\sqrt{3}}{14},\frac{1}{7}\right)=\left(2\sqrt{3},\frac{1}{7}\right)$. Os demais pontos seguem por simetria em relação aos eixos.

\textbf{Resposta.} O lugar de $M$ é a elipse $4x^2+49y^2=49$. Ela intersecta as retas $x=\pm 2\sqrt{3}$ nos pontos
\[
\left(2\sqrt{3},\,\frac{1}{7}\right),\quad \left(2\sqrt{3},\,-\frac{1}{7}\right),\quad \left(-2\sqrt{3},\,\frac{1}{7}\right),\quad \left(-2\sqrt{3},\,-\frac{1}{7}\right).
\]

ANSWER: O lugar de $M$ é a elipse $\dfrac{4x^2}{49}+y^2=1$. Com $c=2\sqrt{3}$, os pontos pedidos são $\left(2\sqrt{3},\pm\frac{1}{7}\right)$ e $\left(-2\sqrt{3},\pm\frac{1}{7}\right)$.
